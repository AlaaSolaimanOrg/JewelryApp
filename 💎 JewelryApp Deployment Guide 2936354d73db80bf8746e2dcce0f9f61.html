🧑‍💻 Deployment Commands

🔄 Pull Latest Code

cd /var/www/myapi
git pull

🧱 Apply Migrations (Production)

cd src/JewerlyApp.API
dotnet ef database update \
  --project ../JewerlyApp.Infrastructure/ \
  --startup-project . \
  --connection "Server=localhost;Database=JewerlyApp;User Id=sa;Password=yU2E3n8VN&B2qiV6y9;TrustServerCertificate=True;"

📦 Publish API

cd src/JewerlyApp.API
dotnet publish -c Release -o /var/www/myapi/src/JewerlyApp.API/publish/ --self-contained false

♻️ Restart API Service

sudo systemctl restart jewerlyapp.service

🪞 Deploy Frontend

cd /var/www/myapi/JewelryApp.UI
rm -rf node_modules package-lock.json
npm i
npm run build:prod

🌐 Nginx Configuration

📝 Edit Config

sudo nano /etc/nginx/sites-available/myapi

✅ Test & Reload

sudo nginx -t
sudo systemctl reload nginx

⚙️ Sample Config

server
{
        listen 80;
        server_name _;

        # Production Frontend
        location /
        {
                root /var/www/myapi/JewelryApp.UI/dist;
                index index.html;
                try_files $uri $uri/ /index.html;
        }

        # Development Frontend
        location /dev/
        {
                alias /var/www/jewelryApp-dev/JewelryApp.UI/dist/;
                index index.html;
                try_files $uri $uri/ /dev/index.html;
        }

        # Production API
        location /api/
        {
                proxy_pass http://127.0.0.1:5000;
                proxy_http_version 1.1;
                proxy_set_header Upgrade $http_upgrade;
                proxy_set_header Connection keep-alive;
                proxy_set_header Host $host;
                proxy_cache_bypass $http_upgrade;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
        }

        # Development API
        location /dev/api/
        {
                proxy_pass http://127.0.0.1:5001/dev/api/;
                proxy_http_version 1.1;
                proxy_set_header Upgrade $http_upgrade;
                proxy_set_header Connection keep-alive;
                proxy_set_header Host $host;
                proxy_cache_bypass $http_upgrade;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
        }

        # Swagger - Development
        location /dev/api/swagger/
        {
                proxy_pass http://127.0.0.1:5001/dev/api/swagger/;
                proxy_http_version 1.1;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
        }
}
